# Form Engine

A high-fidelity Form toolkit engineered for both simple landing pages and complex enterprise dashboards.

This engine decouples layout from state. It provides a stunning visual wrapper with a **slim animated glowing border** that tracks focus, and a custom **glitching error text effect** that draws immediate attention to invalid fields.

## Examples

`FormWrapper` is unopinionated about state — it works with plain `useState` and with `react-hook-form`'s `handleSubmit` alike, because it never clones or intercepts child `ref`s. The first example's **Code** tab contains the full component source.

export const usageNative = `"use client";

import { useState } from "react";
import {
  FormField,
  FormSubmit,
  FormWrapper,
} from "@/components/custom/blocks/FormBlock";
import { InputBlock } from "@/components/custom/blocks/InputBlock";

export default function NewsletterForm() {
  const [email, setEmail] = useState("");
  const [error, setError] = useState("");

  const onSubmit = (e) => {
    e.preventDefault();
    if (!email.includes("@")) {
      setError("Please provide a valid email address.");
      return;
    }
    setError("");
    // Submit logic...
  };

  return (
    <FormWrapper
      title="Newsletter Signup"
      description="A simple, single-field form using native React state."
      onSubmit={onSubmit}
    >
      <FormField label="Email Address" error={error}>
        <InputBlock
          type="email"
          placeholder="Enter your email..."
          value={email}
          onChange={(e) => setEmail(e.target.value)}
        />
      </FormField>

      <FormSubmit>Subscribe Now</FormSubmit>
    </FormWrapper>
  );
}`;

export const usageZod = `"use client";

import { zodResolver } from "@hookform/resolvers/zod";
import { useForm } from "react-hook-form";
import * as z from "zod";
import {
  FormField,
  FormSubmit,
  FormWrapper,
} from "@/components/custom/blocks/FormBlock";
import { InputBlock } from "@/components/custom/blocks/InputBlock";

const schema = z.object({
  companyName: z.string().min(2, "Company name is strictly required."),
  workEmail: z.string().email("Invalid corporate email format detected."),
  password: z.string().min(8, "Security requirement: Minimum 8 characters."),
});

export default function EnterpriseForm() {
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm({ resolver: zodResolver(schema), mode: "onTouched" });

  return (
    <FormWrapper
      title="Enterprise Registration"
      description="Click inside the fields and blur away to trigger Zod validation."
      onSubmit={handleSubmit(async (data) => console.log(data))}
    >
      <div className="grid grid-cols-1 sm:grid-cols-2 gap-6">
        <div className="sm:col-span-2">
          <FormField label="Company Name" error={errors.companyName?.message}>
            <InputBlock {...register("companyName")} placeholder="e.g. Acme Corp" />
          </FormField>
        </div>

        <FormField label="Work Email" error={errors.workEmail?.message}>
          <InputBlock
            {...register("workEmail")}
            type="email"
            placeholder="name@company.com"
          />
        </FormField>

        <FormField label="Secure Password" error={errors.password?.message}>
          <InputBlock {...register("password")} type="password" placeholder="••••••••" />
        </FormField>
      </div>

      <FormSubmit isSubmitting={isSubmitting}>Create Workspace</FormSubmit>
    </FormWrapper>
  );
}`;

export const usageErrorStyles = `import {
  FormField,
  FormWrapper,
} from "@/components/custom/blocks/FormBlock";
import { InputBlock } from "@/components/custom/blocks/InputBlock";

export default function ErrorStyles() {
  const message = "Invalid corporate email format detected.";

  return (
    <FormWrapper
      title="Error Styles"
      description="The same message rendered with each errorVariant."
    >
      {/* "glitch" is the default — animated RGB-split text */}
      <FormField label="Glitch (default)" error={message} errorVariant="glitch">
        <InputBlock type="email" defaultValue="invalid-email" />
      </FormField>

      {/* "default" is the quiet, static variant */}
      <FormField label="Default" error={message} errorVariant="default">
        <InputBlock type="email" defaultValue="invalid-email" />
      </FormField>
    </FormWrapper>
  );
}`;

export const usageSubmitting = `import {
  FormField,
  FormSubmit,
  FormWrapper,
} from "@/components/custom/blocks/FormBlock";
import { InputBlock } from "@/components/custom/blocks/InputBlock";

export default function SubmittingForm() {
  return (
    <FormWrapper
      title="Newsletter Signup"
      description="FormSubmit swaps its label for a spinner while isSubmitting is true."
      onSubmit={(e) => e.preventDefault()}
    >
      <FormField label="Email Address">
        <InputBlock type="email" defaultValue="ada@example.com" />
      </FormField>

      {/* Wire this to react-hook-form's formState.isSubmitting */}
      <FormSubmit isSubmitting>Subscribe Now</FormSubmit>
    </FormWrapper>
  );
}`;

export const usageBare = `import {

  );
}`;

## Installation

<Cli command={`add form.json`} />

The CLI writes the component into your project and installs anything it depends on.


### 1. Copy the Component

This component exports four modular building blocks: `FormWrapper`, `FormField`, `FormError`, and `FormSubmit`.

The high-fidelity styling (the spinning conic gradient and the CSS glitch animations) are self-contained within this file and require no external animation libraries.

Create a new file at `components/custom/blocks/FormBlock.tsx` and copy the engine code from the **Code** tab of the first example above.

### 2. (Optional) Install React Hook Form & Zod

If you intend to use the advanced enterprise validation shown in the second example, you will need to install the following packages:

```bash
npm install react-hook-form zod @hookform/resolvers
```
