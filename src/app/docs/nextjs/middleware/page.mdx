# Middleware

Middleware allows you to run code before a request is completed. You can modify the response, rewrite the request, redirect, and more.

## Creating Middleware

Create a `middleware.js` file at the root of your project:

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  // Your middleware logic here
  return NextResponse.next()
}

export const config = {
  matcher: ['/about/:path*', '/dashboard/:path*'],
}
```

## Basic Example

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  // Add a custom header to the response
  const response = NextResponse.next()
  response.headers.set('x-custom-header', 'my-value')
  return response
}
```

## Authentication

Protect routes with authentication middleware:

```js
// middleware.js
import { NextResponse } from 'next/server'
import { jwtVerify } from 'jose'

export async function middleware(request) {
  const token = request.cookies.get('auth-token')?.value
  
  if (!token) {
    return NextResponse.redirect(new URL('/login', request.url))
  }
  
  try {
    await jwtVerify(
      token,
      new TextEncoder().encode(process.env.JWT_SECRET)
    )
    return NextResponse.next()
  } catch (error) {
    return NextResponse.redirect(new URL('/login', request.url))
  }
}

export const config = {
  matcher: ['/dashboard/:path*', '/profile/:path*'],
}
```

## Redirects

Redirect users based on conditions:

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  const { pathname } = request.nextUrl
  
  // Redirect old paths to new paths
  if (pathname === '/old-page') {
    return NextResponse.redirect(new URL('/new-page', request.url))
  }
  
  // Redirect based on user agent
  const userAgent = request.headers.get('user-agent') || ''
  if (userAgent.includes('Mobile') && pathname === '/desktop-only') {
    return NextResponse.redirect(new URL('/mobile-version', request.url))
  }
  
  return NextResponse.next()
}

export const config = {
  matcher: ['/old-page', '/desktop-only'],
}
```

## Rewrites

Rewrite URLs without changing the browser URL:

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  const { pathname } = request.nextUrl
  
  // Rewrite API calls
  if (pathname.startsWith('/api/external')) {
    return NextResponse.rewrite(
      new URL(`/api/proxy${pathname}`, request.url)
    )
  }
  
  return NextResponse.next()
}

export const config = {
  matcher: ['/api/external/:path*'],
}
```

## Internationalization

Implement internationalization with middleware:

```js
// middleware.js
import { NextResponse } from 'next/server'
import { match } from '@formatjs/intl-localematcher'
import Negotiator from 'negotiator'

const locales = ['en', 'es', 'fr']
const defaultLocale = 'en'

function getLocale(request) {
  const negotiator = new Negotiator({
    headers: { 'accept-language': request.headers.get('accept-language') || '' }
  })
  
  const languages = negotiator.languages()
  return match(languages, locales, defaultLocale)
}

export function middleware(request) {
  const { pathname } = request.nextUrl
  const pathnameHasLocale = locales.some(
    (locale) => pathname.startsWith(`/${locale}/`) || pathname === `/${locale}`
  )
  
  if (pathnameHasLocale) return NextResponse.next()
  
  // Redirect if there is no locale
  const locale = getLocale(request)
  request.nextUrl.pathname = `/${locale}${pathname}`
  return NextResponse.redirect(request.nextUrl)
}

export const config = {
  matcher: [
    // Skip all internal paths (_next)
    '/((?!_next).*)',
    // Optional: only run on root (/) URL
    // '/'
  ],
}
```

## A/B Testing

Implement A/B testing with middleware:

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  // Get existing cookie or set a new one
  let bucket = request.cookies.get('ab-test-bucket')?.value
  
  if (!bucket) {
    // Randomly assign user to a bucket
    bucket = Math.random() > 0.5 ? 'A' : 'B'
  }
  
  const response = NextResponse.next()
  
  // Set cookie if it doesn't exist
  if (!request.cookies.get('ab-test-bucket')) {
    response.cookies.set('ab-test-bucket', bucket, {
      maxAge: 60 * 60 * 24 * 30, // 30 days
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
    })
  }
  
  // Add header for client-side access
  response.headers.set('x-ab-test-bucket', bucket)
  
  return response
}
```

## Rate Limiting

Implement basic rate limiting:

```js
// middleware.js
import { NextResponse } from 'next/server'

// Simple in-memory store (use Redis in production)
const rateLimitStore = new Map()

const RATE_LIMIT = 100 // requests per window
const WINDOW_MS = 60 * 1000 // 1 minute

export function middleware(request) {
  const ip = request.ip || '127.0.0.1'
  const now = Date.now()
  const windowStart = now - WINDOW_MS
  
  // Get or initialize rate limit info
  const rateLimitInfo = rateLimitStore.get(ip) || {
    count: 0,
    lastRequest: now,
  }
  
  // Reset count if outside window
  if (rateLimitInfo.lastRequest < windowStart) {
    rateLimitInfo.count = 0
  }
  
  // Increment count
  rateLimitInfo.count += 1
  rateLimitInfo.lastRequest = now
  rateLimitStore.set(ip, rateLimitInfo)
  
  // Check rate limit
  if (rateLimitInfo.count > RATE_LIMIT) {
    return new NextResponse('Too Many Requests', { status: 429 })
  }
  
  const response = NextResponse.next()
  response.headers.set('x-ratelimit-remaining', (RATE_LIMIT - rateLimitInfo.count).toString())
  return response
}

export const config = {
  matcher: ['/api/:path*'],
}
```

## CORS Handling

Handle CORS with middleware:

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  const response = NextResponse.next()
  
  // Set CORS headers
  response.headers.set('Access-Control-Allow-Origin', '*')
  response.headers.set('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS')
  response.headers.set('Access-Control-Allow-Headers', 'Content-Type, Authorization')
  
  // Handle preflight requests
  if (request.method === 'OPTIONS') {
    return new NextResponse(null, {
      status: 204,
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
        'Access-Control-Allow-Headers': 'Content-Type, Authorization',
      },
    })
  }
  
  return response
}

export const config = {
  matcher: '/api/:path*',
}
```

## Conditional Logic

Apply middleware conditionally based on request properties:

```js
// middleware.js
import { NextResponse } from 'next/server'

export function middleware(request) {
  const { pathname, searchParams } = request.nextUrl
  
  // Skip middleware for static files
  if (pathname.includes('.')) {
    return NextResponse.next()
  }
  
  // Apply different logic based on path
  if (pathname.startsWith('/api')) {
    // API-specific middleware
    return handleApiMiddleware(request)
  }
  
  if (pathname.startsWith('/admin')) {
    // Admin-specific middleware
    return handleAdminMiddleware(request)
  }
  
  return NextResponse.next()
}

function handleApiMiddleware(request) {
  // API-specific logic
  const apiKey = request.headers.get('x-api-key')
  if (!apiKey || apiKey !== process.env.API_KEY) {
    return new NextResponse('Unauthorized', { status: 401 })
  }
  return NextResponse.next()
}

function handleAdminMiddleware(request) {
  // Admin-specific logic
  const authHeader = request.headers.get('authorization')
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return NextResponse.redirect(new URL('/login', request.url))
  }
  return NextResponse.next()
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
}
```

## Best Practices

1. **Keep middleware lightweight** to avoid performance issues
2. **Use the matcher config** to limit which routes middleware runs on
3. **Handle errors gracefully** to prevent application crashes
4. **Avoid heavy computations** in middleware
5. **Use cookies for persistent data** across requests
6. **Implement proper logging** for debugging
7. **Test thoroughly** as middleware affects all matching routes
8. **Use environment variables** for configuration
9. **Consider caching** for expensive operations
10. **Document your middleware** for team understanding
