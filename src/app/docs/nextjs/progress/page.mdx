# Progress

A flexible, smoothly animated progress indicator built to handle loading states, storage meters, or step-by-step wizards.

This block is split into two composable pieces: the core `<Progress />` bar for the visual fill, and the optional `<ProgressField />` wrapper which automatically handles the layout for labels and percentage text.

## Examples

Each example shows a live preview with the exact code to copy. The first example's **Code** tab contains the full component source.

export const usageFormatted = `import { useState } from "react";
import { Progress, ProgressField } from "@/components/custom/blocks/ProgressBlock";

export default function StorageWidget() {
  const [usedSpace] = useState(72); // 72 GB of 100 GB

  return (
    <ProgressField
      label="Storage Usage"
      value={usedSpace}
      max={100}
      valueFormat={(value, max) => \`\${value} GB / \${max} GB\`}
    >
      <Progress value={usedSpace} max={100} variant="glow" barSize="md" />
    </ProgressField>
  );
}`;

export const usageSteps = `import { Progress, ProgressField } from "@/components/custom/blocks/ProgressBlock";

export default function WizardProgress() {
  const step = 2;
  const totalSteps = 4;

  return (
    // showValue={false} hides the percentage when the label already says it.
    <ProgressField
      label={\`Step \${step} of \${totalSteps} — Billing details\`}
      value={step}
      max={totalSteps}
      showValue={false}
    >
      <Progress value={step} max={totalSteps} variant="default" barSize="sm" />
    </ProgressField>
  );
}`;
