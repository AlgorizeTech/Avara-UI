# Avatar

A reusable avatar component with built-in support for single avatars, grouped avatar stacks, sizes, and dynamic online/offline status indicators.

## Single Avatar & Status

Basic avatars with optional status indicators. Use the `active` and `status` props to show user availability.

## Code

```tsx
"use client";

import { Avatar } from "@/components/custom/avatar";

export default function SingleAvatarDemo() {
  const user = [{ src: "[https://github.com/shadcn.png](https://github.com/shadcn.png)", fallback: "CN" }];

  return (
    <div className="flex items-center gap-6">
      <Avatar avatars={user} />

      <Avatar avatars={user} active={true} status="inactive" />
      <Avatar avatars={user} active={true} />
    </div>
  );
}
```

##Code
```tsx
"use client";

import { Avatar } from "@/components/custom/avatar";

export default function GroupAvatarDemo() {
  const team = [
    { src: "[https://github.com/shadcn.png](https://github.com/shadcn.png)", fallback: "CN" },
    { src: "[https://github.com/vercel.png](https://github.com/vercel.png)", fallback: "VC" },
    { src: "[https://github.com/nextjs.png](https://github.com/nextjs.png)", fallback: "NX" },
    { src: "", fallback: "JD" }, // Falls back to initials automatically
    { src: "[https://github.com/react.png](https://github.com/react.png)", fallback: "RC" }
  ];

  return (
    <div className="flex gap-8">
      {/* Standard Group */}
      <Avatar avatars={team.slice(0, 3)} size="default" />
      
      {/* Large Group */}
      <Avatar avatars={team} size="lg" />
    </div>
  );
}
```
