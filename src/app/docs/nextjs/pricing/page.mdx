# Pricing Components

A collection of highly-converting, responsive pricing components. Features include toggleable billing cycles, usage-based sliders, and detailed feature comparison tables.

## Code Usage

```tsx
"use client";
import { BasicPricingPlan } from "@/components/custom/pricing-plan";
import { Rocket, Building } from "lucide-react";

export default function PricingDemo() {
  const data = {
    tabs: [
      {
        id: "monthly",
        label: "Monthly",
        plans: [
          {
            id: "startup",
            title: "Startup",
            subtitle: "Perfect for small teams.",
            price: "$29",
            period: "month",
            buttonText: "Get Started",
            popular: true,
            active: true,
            popularText: "Most popular",
            icon: Rocket, // Inside a standard .tsx file, using raw components is perfectly fine!
            features: [
              "10 projects",
              "Up to 2,000 subscribers",
              "Basic analytics",
              "24-hour support",
            ],
          },
        ],
      },
    ],
  };

  return (
    <div className="w-full">
      <BasicPricingPlan data={data} variant="premium" />
    </div>
  );
}
```
