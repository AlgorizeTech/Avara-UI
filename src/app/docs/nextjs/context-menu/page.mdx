# Context Menu

Displays a menu to the user — such as a set of actions or functions — triggered by a right-click.

Building a Context Menu is difficult because it involves intercepting the browser's native right-click event (`onContextMenu`) and rendering an overlay component precisely at the user's mouse coordinates without it getting clipped by CSS `overflow` properties.

This `ContextMenuBlock` is a **100% dependency-free** solution. It uses React's `createPortal` to escape DOM limitations, and includes highly optimized boundary-collision logic to guarantee the menu never opens off the edge of the user's screen.

## Examples

Right-click (or long-press on touch) inside each dashed box. Try it near the edges — the boundary maths keeps the menu on screen. The first example's **Code** tab contains the full component source.

export const usageDefault = `import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuTrigger,
} from "@/components/custom/blocks/ContextMenuBlock";

export default function RightClickArea() {
  return (
    <ContextMenu>
      {/* This whole element intercepts the right-click */}
      <ContextMenuTrigger className="w-full h-40 border-2 border-dashed rounded-2xl flex items-center justify-center">
        Right-click anywhere inside this box
      </ContextMenuTrigger>

      {/* Rendered in a portal at the cursor position */}
      <ContextMenuContent className="w-64">
        <ContextMenuItem onClick={() => console.log("Edit")}>Edit File</ContextMenuItem>
        <ContextMenuItem onClick={() => console.log("Duplicate")}>Duplicate</ContextMenuItem>
        <ContextMenuItem onClick={() => console.log("Rename")}>Rename</ContextMenuItem>
      </ContextMenuContent>
    </ContextMenu>
  );
}`;

export const usageFull = `import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuLabel,
  ContextMenuSeparator,
  ContextMenuShortcut,
  ContextMenuTrigger,
} from "@/components/custom/blocks/ContextMenuBlock";
import { Copy, Download, FileEdit, Scissors, Share2, Trash } from "lucide-react";

export default function DocumentMenu() {
  const run = (action: string) => console.log(action);

  return (
    <ContextMenu>
      <ContextMenuTrigger className="w-full h-56 border-2 border-dashed rounded-2xl flex items-center justify-center">
        Right-click anywhere inside this box
      </ContextMenuTrigger>

      <ContextMenuContent className="w-64">
        <ContextMenuLabel>Document Actions</ContextMenuLabel>
        <ContextMenuItem onClick={() => run("Edit")}>
          <FileEdit className="w-4 h-4 mr-2" />
          Edit File
          <ContextMenuShortcut>⌘E</ContextMenuShortcut>
        </ContextMenuItem>
        <ContextMenuItem onClick={() => run("Download")}>
          <Download className="w-4 h-4 mr-2" />
          Download PDF
          <ContextMenuShortcut>⌘S</ContextMenuShortcut>
        </ContextMenuItem>
        <ContextMenuItem onClick={() => run("Share")}>
          <Share2 className="w-4 h-4 mr-2" />
          Share Link
        </ContextMenuItem>

        <ContextMenuSeparator />

        <ContextMenuLabel>Clipboard</ContextMenuLabel>
        <ContextMenuItem onClick={() => run("Copy")}>
          <Copy className="w-4 h-4 mr-2" />
          Copy
          <ContextMenuShortcut>⌘C</ContextMenuShortcut>
        </ContextMenuItem>

        {/* disabled dims the row and swallows the click */}
        <ContextMenuItem disabled>
          <Scissors className="w-4 h-4 mr-2" />
          Cut
          <ContextMenuShortcut>⌘X</ContextMenuShortcut>
        </ContextMenuItem>

        <ContextMenuSeparator />

        {/* className restyles a single destructive row */}
        <ContextMenuItem
          onClick={() => run("Delete")}
          className="text-red-600 focus:bg-red-50 focus:text-red-700"
        >
          <Trash className="w-4 h-4 mr-2" />
          Delete
          <ContextMenuShortcut>⌘⌫</ContextMenuShortcut>
        </ContextMenuItem>
      </ContextMenuContent>
    </ContextMenu>
  );
}`;

export const usageInset = `import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuLabel,
  ContextMenuSeparator,
  ContextMenuTrigger,
} from "@/components/custom/blocks/ContextMenuBlock";
import { Download } from "lucide-react";

export default function ViewMenu() {
  const run = (action: string) => console.log(action);

  return (
    <ContextMenu>
      <ContextMenuTrigger className="w-full h-40 border-2 border-dashed rounded-2xl flex items-center justify-center">
        Right-click anywhere inside this box
      </ContextMenuTrigger>

      <ContextMenuContent className="w-64">
        {/* inset pads the left edge so text-only rows line up with icon rows */}
        <ContextMenuLabel inset>View</ContextMenuLabel>
        <ContextMenuItem inset onClick={() => run("List")}>List</ContextMenuItem>
        <ContextMenuItem inset onClick={() => run("Grid")}>Grid</ContextMenuItem>

        <ContextMenuSeparator />

        <ContextMenuItem onClick={() => run("Download")}>
          <Download className="w-4 h-4 mr-2" />
          Download PDF
        </ContextMenuItem>
      </ContextMenuContent>
    </ContextMenu>
  );
}`;
