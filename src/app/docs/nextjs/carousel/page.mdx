# Carousel

A collection of customizable and interactive carousel components for image galleries, hero sections, and fluid presentations.

## Image Carousel

A standard image slider with dot pagination and arrow navigation. Supports auto-play.

## Preview

<div className="w-full max-w-4xl mx-auto my-8 border rounded-xl overflow-hidden shadow-sm">
  <ImageCarousel
    images={[
      "https://images.unsplash.com/photo-1506905925346-21bda4d32df4?w=1920&h=1080&fit=crop",
      "https://images.unsplash.com/photo-1441974231531-c6227db76b6e?w=1920&h=1080&fit=crop",
    ]}
    autoPlay={true}
    autoPlayInterval={3000}
  />
</div>

## Code

```tsx
"use client";

import { ImageCarousel } from "@/components/custom/carousel";

export default function ImageCarouselDemo() {
  const images = [
    "[https://images.unsplash.com/photo-1506905925346-21bda4d32df4?w=1920&h=1080&fit=crop](https://images.unsplash.com/photo-1506905925346-21bda4d32df4?w=1920&h=1080&fit=crop)",
    "[https://images.unsplash.com/photo-1441974231531-c6227db76b6e?w=1920&h=1080&fit=crop](https://images.unsplash.com/photo-1441974231531-c6227db76b6e?w=1920&h=1080&fit=crop)",
  ];

  return (
    <div className="w-full max-w-4xl mx-auto">
      <ImageCarousel
        images={images}
        autoPlay={true}
        autoPlayInterval={3000}
        rootClassName="relative w-full overflow-hidden rounded-xl border shadow-sm"
        slideClassName="w-full h-[400px] object-cover"
      />
    </div>
  );
}
```

```tsx
"use client";

import { CustomCarousel } from "@/components/custom/carousel";

export default function CustomCarouselDemo() {
  const items = [
    {
      img: "[https://images.unsplash.com/photo-1506905925346-21bda4d32df4?w=1920&h=1080&fit=crop](https://images.unsplash.com/photo-1506905925346-21bda4d32df4?w=1920&h=1080&fit=crop)",
      title: "Ocean Paradise",
      des: "Experience the serenity of crystal-clear waters.",
    },
    {
      img: "[https://images.unsplash.com/photo-1441974231531-c6227db76b6e?w=1920&h=1080&fit=crop](https://images.unsplash.com/photo-1441974231531-c6227db76b6e?w=1920&h=1080&fit=crop)",
      title: "Mountain Summit",
      des: "Conquer new heights and witness breathtaking views.",
    },
  ];

  return (
    <div className="w-full rounded-xl overflow-hidden border">
      <CustomCarousel
        items={items}
        imageButtonPosition="bottom"
        rootClassName="flex h-[400px] w-full flex-col items-center justify-center gap-5 bg-cover bg-center"
        thumbClassName="h-12 w-20 bg-black/30 box-content cursor-pointer rounded-lg hover:scale-105 transition-transform"
        activeThumbClassName="border-2 border-white p-0.5 shadow-lg"
      />
    </div>
  );
}
```
