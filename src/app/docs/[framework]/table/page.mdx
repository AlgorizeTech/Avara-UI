# Table

A set of composable table primitives. You write the rows; the component supplies the structure, spacing and theming.

There is **no column config and no required row shape**. Whatever your API returns — nested objects, snake_case keys, `_id` instead of `id` — you map it straight into JSX. Nothing has to be flattened or transformed first.

Every example below is live. Click a sortable header, expand a row, or hover the right edge of a row to reveal its actions.

`Table` is the card. `TableContent` renders the scroll container and the `<table>` element. Everything else maps one-to-one onto the HTML table tags, so `colSpan`, `scope` and any other native attribute pass straight through.

  >
    {status}
  </span>

const money = (value: number) =>
  new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(value);

type SortKey = "reference" | "customer" | "amount" | "issuedAt";

const SORT_ACCESSORS = {
  reference: (row: Invoice) => row.reference,
  customer: (row: Invoice) => row.customer.name,
  amount: (row: Invoice) => row.amount,
  issuedAt: (row: Invoice) => row.issuedAt,
} as const;

export default function SortableTable({
  invoices = sampleInvoices,
}: {
  invoices?: Invoice[];
}) {
  const [sort, setSort] = useState<{
    key: SortKey;
    direction: "asc" | "desc";
  } | null>({ key: "reference", direction: "asc" });

  const toggleSort = (key: SortKey) =>
    setSort((prev) =>
      prev && prev.key === key
        ? { key, direction: prev.direction === "asc" ? "desc" : "asc" }
        : { key, direction: "asc" },
    );

  const sorted = useMemo(() => {
    if (!sort) return invoices;
    const read = SORT_ACCESSORS[sort.key];
    const factor = sort.direction === "asc" ? 1 : -1;
    return [...invoices].sort((a, b) => {
      const av = read(a);
      const bv = read(b);
      if (typeof av === "number" && typeof bv === "number")
        return factor * (av - bv);
      return factor * String(av).localeCompare(String(bv));
    });
  }, [invoices, sort]);

  return (
    <Table>
      <TableHeading
        title="Sortable Invoices"
        description="Click headers to sort column data ascending or descending."
      />

      <TableContent>
        <TableHeader>
          <TableRow isHeader>
            <TableHead
              sortable
              sortDirection={sort?.key === "reference" ? sort.direction : null}
              onSort={() => toggleSort("reference")}
            >
              Invoice
            </TableHead>
            <TableHead
              sortable
              sortDirection={sort?.key === "customer" ? sort.direction : null}
              onSort={() => toggleSort("customer")}
            >
              Customer
            </TableHead>
            <TableHead
              align="right"
              sortable
              sortDirection={sort?.key === "amount" ? sort.direction : null}
              onSort={() => toggleSort("amount")}
            >
              Amount
            </TableHead>
            <TableHead>Status</TableHead>
            <TableHead
              align="right"
              sortable
              sortDirection={sort?.key === "issuedAt" ? sort.direction : null}
              onSort={() => toggleSort("issuedAt")}
            >
              Issued
            </TableHead>
          </TableRow>
        </TableHeader>

        <TableBody>
          {sorted.map((invoice) => (
            <TableRow key={invoice.reference}>
              <TableCell className="font-mono text-xs">
                {invoice.reference}
              </TableCell>

              <TableCell>
                <div className="flex flex-col">
                  <span className="font-semibold text-foreground">
                    {invoice.customer.name}
                  </span>
                  <span className="text-xs text-muted-foreground">
                    {invoice.customer.email}
                  </span>

              </TableCell>

              <TableCell align="right" className="font-medium">
