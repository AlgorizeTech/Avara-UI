# Profile Card

A compact creator profile card with a pointer-reactive pastel aurora cover, pulsing availability dot, scalloped verified seal, expandable skill chips, counting stats, and an animated Follow <DocLink to="button">Button</DocLink>.

export const usageDefault = `import { ProfileCard } from "@/components/custom/blocks/ProfileCardBlock";

export default function ProfileCardExample() {
  return (
    <div className="w-full py-8 flex items-center justify-center">
      <ProfileCard
        variant="aurora"
        name="Nadia Karim"
        role="Product & motion engineer"
        location="United States"
        verified={true}
        active={true}
        onFollowChange={(following) => {
          console.log("Following:", following);
        }}
      />
    </div>
  );
}`;
