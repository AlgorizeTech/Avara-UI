# Profile Card

A compact creator profile card with a pointer-reactive pastel aurora cover, pulsing availability dot, scalloped verified seal, expandable skill chips, counting stats, and an animated Follow <DocLink to="button">Button</DocLink> — alongside an interactive **Account Menu** variant featuring a morphing status pill, workspace switcher, optional storage meter, and two-step sign out.

export const usageDefault = `import { ProfileCard } from "@/components/custom/blocks/ProfileCardBlock";

export default function ProfileCardExample() {
  return (
    <div className="w-full py-8 flex items-center justify-center">
      <ProfileCard
        variant="aurora"
        name="Nadia Karim"
        role="Product & motion engineer"
        location="United States"
        verified={true}
        active={true}
        onFollowChange={(following) => {
          console.log("Following:", following);
        }}
      />
    </div>
  );
}`;

export const usageAccountMenu = `import { AccountMenu } from "@/components/custom/blocks/AccountMenuBlock";

export default function AccountMenuExample() {
  return (
    <div className="w-full py-8 flex items-center justify-center">
      <AccountMenu
        name="John Doe"
        handle="@johndoe"
        showModal={true}
        showStorage={false}
        onStatusChange={(status) => console.log("Status:", status)}
        onWorkspaceChange={(id) => console.log("Workspace:", id)}
      />
    </div>
  );
}`;
