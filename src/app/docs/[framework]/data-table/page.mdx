# Data Table

The same composable primitives as the [Table](../table), plus a state provider that wires up **search, sorting, pagination, row selection and batch actions** for you.

You still write every row yourself, so the data shape stays yours: nested objects, `_id` instead of `id`, values that are computed rather than stored. `DataTable` never asks you to reshape a response before rendering it.

Every example is live. Search for &ldquo;Admin&rdquo;, sort by **Last Active**, or tick a few rows to bring up the floating batch-actions pill.

_Hint: the Edit button in the batch toolbar opens a fully functional custom modal._

## Anatomy

```tsx
<DataTable data={users} getRowId={(u) => u._id}>
  <DataTableHeading title="Users">
    <DataTableSearch />
    <DataTablePageSize />
  </DataTableHeading>

  <DataTableContent>
    <DataTableHeader>
      <DataTableRow isHeader>
        <DataTableHead><DataTableSelectAll /></DataTableHead>
        <DataTableHead sortBy="profile.name">Name</DataTableHead>
      </DataTableRow>
    </DataTableHeader>

    <DataTableBody>
      {(rows) =>                          // already searched, sorted and paged
        rows.map((user) => (
          <DataTableRow key={user._id} row={user}>
            <DataTableCell><DataTableSelect /></DataTableCell>
            <DataTableCell>{user.profile.name}</DataTableCell>
          </DataTableRow>
        ))
      }
    </DataTableBody>
  </DataTableContent>

  <DataTableFooter>
    <DataTableInfo />
    <DataTablePagination />
  </DataTableFooter>

  <DataTableBatchBar>
    {({ selectedRows, clearSelection }) => <button>Delete</button>}
  </DataTableBatchBar>
</DataTable>
```

`DataTableBody` takes a render function and hands you the current page of rows. Everything else is either a layout primitive re-exported from `Table`, or a small wired control that reads the provider.

export const usageBasic = `"use client";

import {
  DataTable,
  DataTableBody,
  DataTableCell,
  DataTableContent,
  DataTableEmpty,
  DataTableFooter,
  DataTableHead,
  DataTableHeader,
  DataTableHeading,
  DataTableInfo,
  DataTablePageSize,
  DataTablePagination,
  DataTableRow,
  DataTableSearch,
} from "@/components/custom/blocks/AdvancedTableBlock";

// Straight off the wire — nested, and keyed by _id.
type User = {
  _id: string;
  profile: { name: string; email: string };
  role: string;
  lastActiveTs: number;
  lastActive: string;
};

export default function UserTable({ users }: { users: User[] }) {
  return (
    <DataTable data={users} getRowId={(user) => user._id}>
      <DataTableHeading title="Team members">
        <DataTableSearch placeholder="Search all columns…" />
        <DataTablePageSize />
      </DataTableHeading>

      <DataTableContent>
        <DataTableHeader>
          <DataTableRow isHeader>
            {/* Dot paths reach into nested objects */}
            <DataTableHead sortBy="profile.name">Name</DataTableHead>
            <DataTableHead sortBy="role">Role</DataTableHead>

            {/* Sort on the timestamp while the cell shows prose */}
            <DataTableHead
              sortBy={(user) => user.lastActiveTs}
              sortId="lastActive"
              sortType="number"
            >
              Last Active
            </DataTableHead>
          </DataTableRow>
        </DataTableHeader>

        <DataTableBody>
          {(rows) =>
            rows.length === 0 ? (
              <DataTableEmpty colSpan={3}>No users found.</DataTableEmpty>
            ) : (
              rows.map((user) => (
                <DataTableRow key={user._id} row={user}>
                  <DataTableCell>{user.profile.name}</DataTableCell>
                  <DataTableCell>{user.role}</DataTableCell>
                  <DataTableCell>{user.lastActive}</DataTableCell>
                </DataTableRow>
              ))
            )
          }
        </DataTableBody>
      </DataTableContent>

      <DataTableFooter>
        <DataTableInfo />
        <DataTablePagination />
      </DataTableFooter>
    </DataTable>
  );
}`;

export const usageSelection = `// Add the two checkbox controls and the floating pill. Selection state,
// select-all-on-page and the indeterminate state are all handled for you.
<DataTableRow isHeader>
  <DataTableHead className="w-11">
    <DataTableSelectAll />
  </DataTableHead>
  <DataTableHead sortBy="name">Name</DataTableHead>
</DataTableRow>

// …and in each body row — DataTableSelect finds its id from the row prop.
<DataTableRow key={user.id} row={user}>
  <DataTableCell className="w-11">
    <DataTableSelect />
  </DataTableCell>
  <DataTableCell>{user.name}</DataTableCell>
</DataTableRow>

<DataTableBatchBar>
  {({ selectedRows, selectedIds, clearSelection }) => (
    <button onClick={() => archive(selectedIds)}>
      Archive {selectedRows.length}
    </button>
  )}
</DataTableBatchBar>`;

export const usageBatch = `// The pill is yours to fill. It floats over the table by default —
// pass floating={false} to place it inline in the heading instead.
<DataTableBatchBar>

      <button onClick={() => exportCsv(selectedIds)}>Export CSV</button>

export const usageSearch = `// By default every value in the row — however deeply nested — is searchable.
<DataTable data={users}>…</DataTable>

// Narrow it when you only want certain fields to match:
<DataTable
  data={users}
  getSearchText={(user) => user.profile.name + " " + user.profile.email}
>
  …
</DataTable>`;

export const usageHook = `// Everything the wired controls use is available to you directly.
import { useDataTable } from "@/components/custom/blocks/AdvancedTableBlock";

  return (
    <span>
      {total} matching · {selectedRows.length} selected
    </span>
  );
}

// Render it anywhere inside <DataTable>.`;

## Best Practices & Guidelines

<div className="flex flex-col gap-4 my-6 not-prose">
  <div className="p-5 rounded-xl border border-blue-200 dark:border-blue-900/50 bg-blue-50/50 dark:bg-blue-950/20">
    <h4 className="text-base font-semibold text-blue-900 dark:text-blue-300 mb-1">Give rows a stable id</h4>
    <p className="text-blue-800 dark:text-blue-400/80 text-sm">Selection is tracked by id, not by index, so it survives sorting and paging. If your rows have no natural key, derive one before passing the array in — an array index will unselect the wrong row as soon as the order changes.</p>
  </div>
  <div className="p-5 rounded-xl border border-amber-200 dark:border-amber-900/50 bg-amber-50/50 dark:bg-amber-950/20">
    <h4 className="text-base font-semibold text-amber-900 dark:text-amber-300 mb-1">Client-side by design</h4>
    <p className="text-amber-800 dark:text-amber-400/80 text-sm">Search, sort and paging all run over the array you pass in. For server-driven datasets, use the plain Table primitives and drive the query yourself — or keep DataTable for the current page only and supply your own footer.</p>
  </div>
  <div className="p-5 rounded-xl border border-emerald-200 dark:border-emerald-900/50 bg-emerald-50/50 dark:bg-emerald-950/20">
    <h4 className="text-base font-semibold text-emerald-900 dark:text-emerald-300 mb-1">Give a function sortBy an id</h4>
    <p className="text-emerald-800 dark:text-emerald-400/80 text-sm">A string sortBy doubles as its own identity. When you pass an accessor function, add sortId so the active-sort indicator stays stable across re-renders.</p>
  </div>
</div>
