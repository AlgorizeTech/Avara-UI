# Command Palette

A fast, keyboard-first Command Palette with a search box trigger, instant filtering, hotkey badges, grouped sections, and a modal overlay.

Starts as a compact search bar (`Search anything... [Ctrl] [K]`) in your header. Clicking it or pressing `Ctrl+K` / `⌘K` opens the palette as a centered modal over a dimmed overlay. Pass the variant name via the `variant` prop (`variant="linear"`), ready to support more variants as they are added.

export const usageDefault = `"use client";

import { CommandPalette } from "@/components/custom/blocks/CommandPaletteBlock";

export default function AppHeader() {
  return (
    <header className="flex items-center justify-between p-4 border-b">
      <span className="text-sm font-medium">Projects</span>

      {/* Renders the search box trigger initially; opens modal on click or Ctrl+K */}
      <CommandPalette
        variant="linear"
        triggerPlaceholder="Search anything..."
        placeholder="Search pages, projects and actions"
        onSelect={(item) => console.log("Selected:", item.id)}
      />
    </header>
  );
}`;
