# Sidebar

A powerful, composable sidebar component that automatically manages mobile flyout states and desktop collapse logic.

Building a good sidebar is one of the hardest parts of dashboard development. It requires managing state across completely different structural elements, handling viewport resizing, and animating complex CSS properties smoothly. This `SidebarBlock` utilizes React Context to handle all of this internally.

Click the collapse trigger in each preview's header to watch the labels hide and the icons centre — hover a collapsed icon and its label appears in a <DocLink to="tooltip">Tooltip</DocLink>. Switch a preview to the phone size and the sidebar becomes a dismissible flyout instead.

export const usageDefault = `"use client";

import {
  MobileSidebarTrigger,
  Sidebar,
  SidebarBrand,
  SidebarContent,
  SidebarFooter,
  SidebarGroup,
  SidebarHeader,
  SidebarItem,
  SidebarProvider,
  SidebarTrigger,
} from "@/components/custom/blocks/SidebarBlock";
import { CreditCard, Hexagon, LayoutDashboard, LogOut, Settings, Users } from "lucide-react";
import { useState } from "react";

export default function AppLayout({ children }: { children: React.ReactNode }) {
  const [activeRoute, setActiveRoute] = useState("Dashboard");

  return (
    // The provider owns expanded/mobile state for every part below it.
    <SidebarProvider defaultExpanded>
      <Sidebar variant="attached">
        <SidebarHeader>
          <SidebarBrand icon={<Hexagon className="w-6 h-6 text-primary" fill="currentColor" />}>
            Nexus UI
          </SidebarBrand>
        </SidebarHeader>

        <SidebarContent>
          {/* SidebarGroup labels fade out when the rail collapses */}
          <SidebarGroup label="Platform">
            <SidebarItem
              icon={<LayoutDashboard className="w-5 h-5" />}
              active={activeRoute === "Dashboard"}
              onClick={() => setActiveRoute("Dashboard")}
            >
              Dashboard
            </SidebarItem>
            <SidebarItem
              icon={<Users className="w-5 h-5" />}
              active={activeRoute === "Customers"}
              onClick={() => setActiveRoute("Customers")}
            >
              Customers
            </SidebarItem>
            <SidebarItem icon={<CreditCard className="w-5 h-5" />}>Billing</SidebarItem>
          </SidebarGroup>

          <SidebarGroup label="Support">
            <SidebarItem icon={<Settings className="w-5 h-5" />}>Settings</SidebarItem>
          </SidebarGroup>
        </SidebarContent>

        <SidebarFooter>
          <SidebarItem icon={<LogOut className="w-5 h-5" />}>Log out</SidebarItem>
        </SidebarFooter>
      </Sidebar>

      <div className="flex-1 flex flex-col min-w-0">
        <header className="h-16 flex items-center gap-4 px-6 border-b">
          {/* Opens the flyout under md; collapses the rail above it */}
          <MobileSidebarTrigger className="md:hidden" />
          <SidebarTrigger />
          <h1 className="text-lg font-bold">{activeRoute}</h1>
        </header>

        <main className="flex-1 p-6 overflow-y-auto">{children}</main>
      </div>
    </SidebarProvider>
  );
}`;

export const usageCollapsed = `import {

export default function RailLayout({ children }: { children: React.ReactNode }) {
  return (
    // Start as an icon rail — labels appear as tooltips on hover.
    <SidebarProvider defaultExpanded={false}>
      <Sidebar variant="attached">
        <SidebarContent>

            <SidebarItem icon={<LayoutDashboard className="w-5 h-5" />} active>

      <div className="flex-1 flex flex-col min-w-0">

          <SidebarTrigger />
        </header>
        <main className="flex-1 p-6">{children}</main>
      </div>
    </SidebarProvider>
  );
}`;
