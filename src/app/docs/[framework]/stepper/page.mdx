# Stepper

A production-grade, highly animated multi-step wizard component engineered exclusively with **GSAP**. It splits the layout into two decoupled blocks: the visual progress tracker and the bidirectional sliding content payload.

Unlike basic UI placeholders, this engine is designed to handle **real form state, per-step input validation, and asynchronous API submission handling** (including error states) with silky 60fps transitions.

`Stepper` (the tracker) and `StepperContent` (the sliding payload) are decoupled — they only share the `steps` array and a `currentStep` number that you own, so `Stepper` also works alone as a progress header above content you lay out yourself. Validation, submission and error handling stay in your component.

export const usageFlow = `"use client";

import {
  AlertCircle,
  Building2,
  Check,
  CheckCircle2,
  ChevronLeft,
  ChevronRight,
  Loader2,
  Lock,
  Mail,
  ShieldCheck,
  User,
  XCircle,
  type LucideIcon,
} from "lucide-react";
import { useState } from "react";
import { cn } from "@/lib/utils";
import {
  Stepper,
  StepperContent,
} from "@/components/custom/blocks/StepperBlock";
// The Workspace step picks a role with the Select block:
//   npx shadcn@latest add https://algorize-ui.vercel.app/r/select.json
import { SelectBlock } from "@/components/custom/blocks/SelectBlock";

const PRIMARY_ROLE_OPTIONS = [
  "Developer",
  "Designer",
  "Product Manager",
  "Founder",
].map((r) => ({ label: r, value: r }));

const BACK_BTN =
  "flex items-center gap-1.5 px-4 py-2.5 rounded-xl border border-zinc-200 dark:border-zinc-800 bg-zinc-100 dark:bg-zinc-900 text-zinc-700 dark:text-zinc-300 font-semibold text-sm disabled:opacity-40 disabled:cursor-not-allowed hover:bg-zinc-200/60 dark:hover:bg-zinc-800 transition-all active:scale-95 cursor-pointer";
const NEXT_BTN =
  "flex items-center gap-1.5 px-6 py-2.5 rounded-xl text-white font-bold text-sm shadow-md transition-all hover:brightness-110 active:scale-95 cursor-pointer bg-indigo-600";

// Controlled field. \`error\` is what makes a blocked Next visible to the reader.
function Field({
  label,
  icon: Icon,
  placeholder,
  value,
  onChange,
  error,
  type,
}: {
  label: string;
  icon?: LucideIcon;
  placeholder: string;
  value: string;
  onChange: (next: string) => void;
  error?: string;
  type?: string;
}) {
  return (
    <label className="flex flex-col gap-1.5 w-full">
      <span className="text-xs font-bold text-zinc-700 dark:text-zinc-300 uppercase tracking-wider">
        {label} <span className="text-rose-500">*</span>
      </span>
      <div className="relative flex items-center">
        {Icon && (
          <div className="absolute left-3.5 text-zinc-400 dark:text-zinc-500 pointer-events-none">
            <Icon className="w-4 h-4" />
          </div>
        )}
        <input
          type={type}
          value={value}
          onChange={(e) => onChange(e.target.value)}
          placeholder={placeholder}
          className={cn(
            "w-full h-11 bg-zinc-50 dark:bg-zinc-900 border rounded-xl px-3.5 text-sm text-zinc-900 dark:text-white placeholder:text-zinc-400 focus:outline-none focus:ring-2 transition-all",
            Icon && "pl-10",
            error
              ? "border-rose-500 focus:ring-rose-500/20"
              : "border-zinc-200 dark:border-zinc-800 focus:border-zinc-400 dark:focus:border-zinc-600",
          )}
        />

      {error && (

// Every step body sits on this card — it is what the preview renders.
function StepCard({
  title,
  subtitle,
  children,
}: {
  title: string;
  subtitle: string;
  children: React.ReactNode;
}) {
  return (
    <div className="w-full p-6 sm:p-8 bg-white dark:bg-[#0a0a0a] rounded-2xl border border-zinc-200 dark:border-zinc-800 shadow-sm flex flex-col gap-5">
      <div>
        <h3 className="text-lg font-bold text-zinc-900 dark:text-white">{title}</h3>
        <p className="text-xs text-zinc-500">{subtitle}</p>
      </div>
      {children}
    </div>
  );
}

type FormData = {
  fullName: string;
  email: string;
  company: string;
  role: string;
  password: string;
  confirmPassword: string;
};
type Errors = Partial<Record<keyof FormData, string>>;

export default function OnboardingWizard() {
  const [step, setStep] = useState(0);
  const [formData, setFormData] = useState<FormData>({
    fullName: "",
    email: "",
    company: "",
    role: "Developer",
    password: "",
    confirmPassword: "",
  });
  const [errors, setErrors] = useState<Errors>({});
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [simulateError, setSimulateError] = useState(false);
  const [submitError, setSubmitError] = useState("");

  const set = (key: keyof FormData) => (next: string) =>
    setFormData((prev) => ({ ...prev, [key]: next }));

  // Returns false AND records the reason, so the blocked step explains itself.
  const validateStep = (current: number) => {
    const next: Errors = {};
    if (current === 0 && !formData.fullName) next.fullName = "Name required";
    if (current === 1 && !formData.company) next.company = "Company required";
    if (current === 2 && !formData.password) next.password = "Password required";
    setErrors(next);
    return Object.keys(next).length === 0;
  };

  const handleNext = () => {
    if (validateStep(step)) setStep((s) => s + 1);
  };

  const handleSubmit = async () => {
    setIsSubmitting(true);
    setSubmitError("");
    try {
      // simulateError is the preview's DEBUG_MODE switch — it lets you see the
      // failure branch without breaking the endpoint. Drop it in your own app.
      if (simulateError) throw new Error("ERR_500: Database insertion failed.");
      await fetch("/api/submit", {
        method: "POST",
        body: JSON.stringify(formData),
      });
    } catch (e) {
      setSubmitError(e instanceof Error ? e.message : "Submission failed");
    } finally {
      setIsSubmitting(false);
    }
  };

  const steps = [
    {
      id: "profile",
      title: "Profile",
      description: "Personal details",
      icon: User,
      content: (
        <StepCard title="Personal Information" subtitle="Enter your contact details to begin setting up your account.">
          <div className="grid gap-4 sm:grid-cols-2">
            <Field
              label="Full Name"
              icon={User}
              placeholder="e.g. Alex Rivera"
              value={formData.fullName}
              onChange={set("fullName")}
              error={errors.fullName}
            />
            <Field
              label="Work Email"
              icon={Mail}
              type="email"
              placeholder="alex@company.com"
              value={formData.email}
              onChange={set("email")}
            />
          </div>
        </StepCard>
      ),
    },
    {
      id: "workspace",
      title: "Workspace",
      description: "Organization",
      icon: Building2,
      content: (
        <StepCard title="Workspace Setup" subtitle="Name your organization and select your primary role.">
          <div className="grid gap-4 sm:grid-cols-2">
            <Field
              label="Company Name"
              icon={Building2}
              placeholder="e.g. Acme Corp"
              value={formData.company}
              onChange={set("company")}
              error={errors.company}
            />
            <div className="flex flex-col gap-1.5 w-full">
              <label className="text-xs font-bold text-zinc-700 dark:text-zinc-300 uppercase tracking-wider">
                Primary Role
              </label>
              <SelectBlock
                options={PRIMARY_ROLE_OPTIONS}
                value={formData.role}
                onChange={set("role")}
                variant="filled"
                placeholder="Select your role"
              />
            </div>
          </div>
        </StepCard>
      ),
    },
    {
      id: "security",
      title: "Security",
      description: "Credentials",
      icon: ShieldCheck,
      content: (
        <StepCard title="Security Credentials" subtitle="Create a secure password for your workspace access.">
          <div className="grid gap-4 sm:grid-cols-2">
            <Field
              label="Password"
              icon={Lock}
              type="password"
              placeholder="Min 6 characters"
              value={formData.password}
              onChange={set("password")}
              error={errors.password}
            />
            <Field
              label="Confirm Password"
              icon={Lock}
              type="password"
              placeholder="Re-type password"
              value={formData.confirmPassword}
              onChange={set("confirmPassword")}
            />
          </div>
        </StepCard>
      ),
    },
    {
      id: "review",
      title: "Review",
      description: "Confirm & Submit",
      icon: CheckCircle2,
      content: (
        <StepCard title="Review Summary" subtitle="Please verify your details before submitting the onboarding payload.">
          <div className="grid grid-cols-2 gap-4 bg-zinc-50 dark:bg-zinc-900/50 p-4 rounded-xl border border-zinc-200 dark:border-zinc-800 text-xs">
            <div>
              <span className="text-zinc-400 block font-semibold">Name:</span>
              <span className="font-bold text-zinc-900 dark:text-white text-sm">{formData.fullName || "—"}</span>
            </div>
            <div>
              <span className="text-zinc-400 block font-semibold">Email:</span>
              <span className="font-bold text-zinc-900 dark:text-white text-sm truncate block">{formData.email || "—"}</span>
            </div>
            <div>
              <span className="text-zinc-400 block font-semibold">Company:</span>
              <span className="font-bold text-zinc-900 dark:text-white text-sm">{formData.company || "—"}</span>
            </div>
            <div>
              <span className="text-zinc-400 block font-semibold">Role:</span>
              <span className="font-bold text-zinc-900 dark:text-white text-sm">{formData.role || "—"}</span>
            </div>
          </div>

          {submitError && (
            <div className="p-4 bg-rose-500/10 border border-rose-500/30 rounded-xl flex items-start gap-3 text-rose-500">
              <XCircle className="w-5 h-5 flex-shrink-0 mt-0.5" />
              <div className="text-xs">
                <strong className="font-bold block mb-0.5">Submission Failed</strong>
                <span>{submitError}</span>
              </div>
            </div>
          )}

          <div className="pt-2 border-t border-zinc-100 dark:border-zinc-800/80 flex items-center justify-between">
            <label className="flex items-center gap-2 cursor-pointer select-none text-xs font-semibold text-zinc-600 dark:text-zinc-400">
              <input
                type="checkbox"
                checked={simulateError}
                onChange={(e) => setSimulateError(e.target.checked)}
                className="rounded border-zinc-300 dark:border-zinc-700 text-rose-500 focus:ring-rose-500"
              />
              Simulate Server Error on Submit
            </label>
            <span className="text-[10px] text-zinc-400 font-mono">DEBUG_MODE</span>
          </div>
        </StepCard>
      ),
    },
  ];

  return (
    <div className="max-w-3xl mx-auto flex flex-col gap-8">
      {/* onStepChange is what makes the circles clickable — gate it on validation */}
      <Stepper
        steps={steps}
        currentStep={step}
        onStepChange={(target) => {
          if (target < step || validateStep(step)) setStep(target);
        }}
      />

      <StepperContent steps={steps} currentStep={step} />

      <div className="flex items-center justify-between pt-6 border-t border-zinc-100 dark:border-zinc-900">
        <button
          type="button"
          onClick={() => setStep((s) => s - 1)}
          disabled={step === 0 || isSubmitting}
          className={BACK_BTN}
        >
          <ChevronLeft className="w-4 h-4" />
          Back
        </button>

        {step === steps.length - 1 ? (
          <button
            type="button"
            onClick={handleSubmit}
            disabled={isSubmitting}
            className={NEXT_BTN}
          >
            {isSubmitting ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                <span>Submitting...</span>
              </>
            ) : (
              <>
                <span>Submit Onboarding</span>
                <Check className="w-4 h-4 stroke-[3]" />
              </>
            )}
          </button>
        ) : (
          <button type="button" onClick={handleNext} className={NEXT_BTN}>
            <span>Next Step</span>
            <ChevronRight className="w-4 h-4" />
          </button>
        )}
      </div>
    </div>
  );
}`;
