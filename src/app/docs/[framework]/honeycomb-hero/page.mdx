# Honeycomb Hero

A nature or travel hero: a light line over a bold, tracked accent word on the left, and on the right a honeycomb of five hexagonal photographs with a hexagonal call to action set among them. Each cell lifts a touch on hover; a row of social links runs along the foot over a faint grid that fades towards the edges.

The hexagons are a single `clip-path` polygon on each cell, so the photographs stay ordinary `<img>` elements — swap any of the five for your own.

export const usage = `"use client";

import Link from "next/link";
import { HoneycombHero } from "@/components/custom/blocks/HoneycombHeroBlock";

const IMAGES = [
  { src: "/norway/waterfall.jpg", alt: "A foggy waterfall in a green valley" },
  { src: "/norway/forest.jpg", alt: "A sunlit forest path" },
  { src: "/norway/fjord.jpg", alt: "An emerald fjord beach" },
  { src: "/norway/trees.jpg", alt: "Sunlight through wilderness trees" },
  { src: "/norway/peak.jpg", alt: "An alpine peak reflected in a lake" },
];

export default function Hero() {
  return (
    <HoneycombHero
      title="Amazing"
      titleAccent="Norway"
      subtitle="Visit Norway, you will never regret it. Fantastic nature, fjords, and the fancy outlines of the coast."
      primaryActionLabel="Book a tour"
      primaryActionHref="/tours"
      images={IMAGES}
      socialLinks={[
        { label: "Instagram", href: "https://instagram.com/yourpage" },
        { label: "YouTube", href: "https://youtube.com/yourchannel" },
      ]}
      color="#d4a017"
      linkComponent={Link}
    />
  );
}
`;
