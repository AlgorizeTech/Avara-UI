# Avatar Stack

A row of overlapping faces — the "people who…" cue under a headline, beside a count, or next to a join button. It shows up to `max` faces, folds the rest into a "+N" disc, and takes an optional trailing `action` that shares the stack's size. On hover the faces fan apart a little, so the row answers the pointer without turning into a carousel.

Several sections used to hand-roll this row; <DocLink to="fashion-hero">Fashion Hero</DocLink> uses this one.

import { AvatarStack } from "@/components/custom/blocks/AvatarStackBlock";

const PEOPLE = [

  { src: "/people/tom.jpg", alt: "Tom" },
  { src: "/people/mara.jpg", alt: "Mara" },
];

export default function Community() {
  return (
    <AvatarStack
      avatars={PEOPLE}
      size="md"
      action={
        <a href="/join" aria-label="Join the community" className="grid size-full place-items-center rounded-full bg-[#111] text-white ring-2 ring-white">
          <Plus size={16} />
        </a>
      }
    />
  );
}`;
