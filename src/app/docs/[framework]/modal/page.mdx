# Modal Dialog

A highly animated, dynamic Modal component. Built from scratch using `gsap` for butter-smooth scaling and fading animations. It includes distinct visual variants and content types to handle alerts, forms, and complex interfaces.

Click the trigger in each preview to open that modal. The `type` prop swaps the layout inside the modal; `variant` restyles the container.

_Note: in these previews the modal is absolutely positioned so it stays inside the box. In your app it covers the whole viewport._

export const usageDefault = `import { Modal, ModalHeader, ModalBody, ModalFooter } from "@/components/custom/blocks/ModalBlock";
import { Mail, BadgeCheck } from "lucide-react";

type User = {
  name?: string;
  email?: string;
  emailVerified?: boolean;
  phone?: string;
  address?: string;
};

type UserDetailsModalProps = {
  data?: User | null;
  isOpen: boolean;
  onClose: () => void;
};

export default function UserDetailsModal({ data, isOpen, onClose }: UserDetailsModalProps) {
  return (
    <Modal isOpen={isOpen} onClose={onClose} contentClassName="max-w-2xl">
      <ModalHeader title="User Details" className="border-b border-border" />
      <ModalBody>
        {data && (
          <div className="divide-y divide-border space-y-4">
            {/* Identity */}
            <div className="flex items-start gap-4 pb-5">
              <div className="relative shrink-0">
                <div className="flex h-16 w-16 items-center justify-center rounded-full bg-primary text-white font-bold">
                  {data.name?.[0] || "U"}
                </div>
                {data.emailVerified && (
                  <BadgeCheck className="absolute -bottom-0.5 -right-0.5 h-5 w-5 fill-green-500 text-white" />
                )}
              </div>
              <div className="min-w-0 flex-1">
                <h2 className="text-lg font-semibold">{data.name || "Unnamed user"}</h2>
                <div className="mt-1 flex items-center gap-1.5 text-sm text-muted-foreground">
                  <Mail className="h-3.5 w-3.5" />
                  <span>{data.email || "—"}</span>
                </div>
              </div>
            </div>

            {/* Contact */}
            <div className="pt-4 grid grid-cols-1 sm:grid-cols-2 gap-4 text-sm">
              <div><span className="font-semibold">Phone:</span> {data.phone || "—"}</div>
              <div><span className="font-semibold">Address:</span> {data.address || "—"}</div>
            </div>
          </div>
        )}
      </ModalBody>
      <ModalFooter className="border-t border-border">
        <button onClick={onClose} className="px-4 py-2 bg-secondary text-secondary-foreground rounded-md">
          Close
        </button>
      </ModalFooter>
    </Modal>
  );
}`;
