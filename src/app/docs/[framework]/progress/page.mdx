# Progress

A flexible, smoothly animated progress indicator built to handle loading states, storage meters, or step-by-step wizards.

This block is split into composable pieces: the core `<Progress />` bar for the visual fill, the optional `<ProgressField />` wrapper which handles the layout for labels and percentage text, `<MultiProgress />` with `<ProgressSegment />` children for several values sharing one track, and `<CircularProgress />` for the same value drawn as a ring.

Each example shows a live preview with the exact code to copy.

export const usageMulti = `import {
  MultiProgress,
  ProgressField,
  ProgressSegment,
} from "@/components/custom/blocks/ProgressBlock";

const phases = [
  { label: "Merged", value: 30 },
  { label: "In review", value: 18 },
  { label: "Drafted", value: 11 },
];

export default function ReleaseProgress() {
  const shipped = phases.reduce((sum, phase) => sum + phase.value, 0);

  return (
    <ProgressField
      label="Release 2.4 progress"
      value={shipped}
      max={100}
      valueFormat={(value, max) => \`\${value} / \${max} tasks\`}
    >
      {/* gap pulls the segments apart so each level reads on its own. */}
      <MultiProgress max={100} gap={3}>
        {phases.map((phase) => (
          <ProgressSegment
            key={phase.label}
            value={phase.value}
            label={\`\${phase.label}: \${phase.value}\`}
          />
        ))}
      </MultiProgress>
    </ProgressField>
  );
}`;

export const usageMultiColors = `import {
  MultiProgress,
  ProgressField,
  ProgressSegment,
  getThemeStyles,
} from "@/components/custom/blocks/ProgressBlock";

const segments = [
  { label: "Documents", value: 18, color: "indigo" },
  { label: "Photos", value: 34, color: "emerald" },
  { label: "Videos", value: 21, color: "rose" },
  { label: "System", value: 9, color: "#f59e0b" },
];

// The legend reuses the block's theme table, so a dot and its segment
// can never drift apart.
function LegendDot({ color }) {
  const { theme, style } = getThemeStyles(color);
  return (
    <span
      className={\`inline-block size-2.5 rounded-full \${theme.bg}\`}
      style={style}
    />
  );
}

export default function StorageBreakdown() {
  const used = segments.reduce((sum, segment) => sum + segment.value, 0);

  return (
    <ProgressField
      label="Storage breakdown"
      value={used}
      max={100}
      valueFormat={(value, max) => \`\${value} GB / \${max} GB\`}
    >
      <MultiProgress max={100}>
        {segments.map((segment) => (
          <ProgressSegment
            key={segment.label}
            value={segment.value}
            color={segment.color}
            label={\`\${segment.label}: \${segment.value} GB\`}
          />
        ))}
      </MultiProgress>

      <div className="flex flex-wrap gap-x-4 gap-y-1.5 pt-1">
        {segments.map((segment) => (
          <span
            key={segment.label}
            className="flex items-center gap-2 text-xs text-muted-foreground"
          >
            <LegendDot color={segment.color} />
            {segment.label}
          </span>
        ))}
      </div>
    </ProgressField>
  );
}`;
